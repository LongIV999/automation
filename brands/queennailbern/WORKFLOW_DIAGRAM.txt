┌─────────────────────────────────────────────────────────────────────────┐
│  CHECK GOOGLE SHEETS                                                     │
│  ─────────────────────────────────────────────────────────────────────  │
│  Query: Content_Calendar tab                                             │
│  Filter:                                                                 │
│    WHERE Status = "scheduled"                                            │
│    AND Date = TODAY                                                      │
│    AND Time = CURRENT_HOUR                                               │
│                                                                           │
│  If found → Continue                                                     │
│  If not   → Stop (wait for next hour)                                    │

                                      │
                                      ▼
┌─────────────────────────────────────────────────────────────────────────┐
│  FACEBOOK GRAPH API: Upload Photos                                      │

│    POST /v19.0/{page-id}/photos                                         │
│    {                                                                    │
│      "url": "image_url",                                                │
│      "published": false,  // Unpublished (for carousel)                │
│      "access_token": "..."                                              │
│    }                                                                    │
│                                                                           │
│  Collect photo IDs: [photo1_id, photo2_id, ..., photo7_id]             │
└─────────────────────────────────────────────────────────────────────────┘
                                      │
                                      ▼

│  {                                                                       │
│    "message": "Caption from Sheet",                                     │
│    "attached_media": [                                                  │
│      {"media_fbid": "photo1_id"},                                       │
│      {"media_fbid": "photo2_id"},                                       │
│      ...                                                                 │
│      {"media_fbid": "photo7_id"}                                        │
│    ],                                                                    │
│    "access_token": "..."                                                │
│  }                                                                       │
│                                                                           │
│  Response: { "id": "POST_ID" }                                          │
